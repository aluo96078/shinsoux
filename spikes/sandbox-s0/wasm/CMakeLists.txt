cmake_minimum_required(VERSION 3.20)
project(shinsou_sandbox_wasm C)

# Standalone: does not touch the repo-root spike CMakeLists. Configure once per flavour:
#   cmake -S wasm -B build/wasm-fast    -DWAMR_FAST_INTERP=1
#   cmake -S wasm -B build/wasm-classic -DWAMR_FAST_INTERP=0
#   cmake -S wasm -B build/wasm-native  -DSW_BACKEND=native
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG")

set(S0 ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(QJS_DIR ${S0}/third_party/quickjs)
set(SW_BACKEND "wamr" CACHE STRING "wamr | native | w2c")
set(WAMR_FAST_INTERP 1 CACHE STRING "1 = fast interpreter, 0 = classic interpreter")
set(WAMR_BLOCK_CACHE 0 CACHE STRING "classic interpreter block-address cache entries (power of 2; 0 = WAMR default 64)")
set(WAMR_THREAD_MGR 1 CACHE STRING "1 = enable WAMR thread manager (terminate checks in interpreter)")
find_package(Threads REQUIRED)

if(SW_BACKEND STREQUAL "native")
  add_library(quickjs STATIC ${QJS_DIR}/dtoa.c ${QJS_DIR}/libregexp.c ${QJS_DIR}/libunicode.c ${QJS_DIR}/quickjs.c)
  target_include_directories(quickjs PUBLIC ${QJS_DIR})
  target_compile_definitions(quickjs PRIVATE _GNU_SOURCE)
  target_compile_options(quickjs PRIVATE -w)
  add_library(core STATIC ${S0}/core/src/shinsou_sandbox_core.c)
  target_include_directories(core PUBLIC ${S0}/core/src)
  target_link_libraries(core PUBLIC quickjs)
  set(BACKEND_SRC backend_native.c)
  set(BACKEND_LIBS core)
  set(FLAVOUR native)
elseif(SW_BACKEND STREQUAL "w2c")
  # Reference only: guest translated to C with wasm2c (needs wabt: brew install wabt).
  set(WABT_PREFIX /opt/homebrew/opt/wabt CACHE PATH "wabt install prefix")
  set(W2C_GEN ${CMAKE_BINARY_DIR}/w2c-gen)
  set(W2C_OUTS)
  foreach(i RANGE 0 7)
    list(APPEND W2C_OUTS ${W2C_GEN}/sandbox_${i}.c)
  endforeach()
  add_custom_command(
    OUTPUT ${W2C_OUTS} ${W2C_GEN}/sandbox.h
    COMMAND ${CMAKE_COMMAND} -E make_directory ${W2C_GEN}
    COMMAND ${WABT_PREFIX}/bin/wasm2c $ENV{SW_WASM} -n sandbox --num-outputs 8 -o ${W2C_GEN}/sandbox.c
    DEPENDS $ENV{SW_WASM})
  set(W2C_RT ${WABT_PREFIX}/share/wabt/wasm2c)
  add_library(w2c STATIC ${W2C_OUTS} ${W2C_RT}/wasm-rt-impl.c ${W2C_RT}/wasm-rt-mem-impl.c ${W2C_RT}/wasm-rt-exceptions-impl.c)
  target_include_directories(w2c PUBLIC ${W2C_GEN} ${WABT_PREFIX}/include ${W2C_RT})
  target_compile_definitions(w2c PUBLIC WASM_RT_USE_MMAP=0 WASM_RT_MEMCHECK_BOUNDS_CHECK=1 WASM_RT_MAX_CALL_STACK_DEPTH=100000)
  target_compile_options(w2c PRIVATE -w -O2)
  set(BACKEND_SRC backend_w2c.c frame_host.c)
  set(BACKEND_LIBS w2c)
  set(FLAVOUR wasm2c)
else()
  set(WAMR_ROOT_DIR ${S0}/third_party/wamr)
  set(WAMR_BUILD_PLATFORM "darwin")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(WAMR_BUILD_TARGET "AARCH64")
  else()
    set(WAMR_BUILD_TARGET "X86_64")
  endif()
  set(WAMR_BUILD_INTERP 1)
  set(WAMR_BUILD_FAST_INTERP ${WAMR_FAST_INTERP})
  set(WAMR_BUILD_AOT 0)
  set(WAMR_BUILD_JIT 0)
  set(WAMR_BUILD_FAST_JIT 0)
  set(WAMR_BUILD_LIBC_BUILTIN 0)
  set(WAMR_BUILD_LIBC_WASI 0)
  set(WAMR_BUILD_MULTI_MODULE 0)
  set(WAMR_BUILD_LIB_PTHREAD 0)
  set(WAMR_BUILD_SHARED_MEMORY 0)
  set(WAMR_BUILD_THREAD_MGR ${WAMR_THREAD_MGR})
  set(WAMR_BUILD_BULK_MEMORY 1)
  set(WAMR_BUILD_REF_TYPES 1)
  set(WAMR_BUILD_SIMD 0)
  set(WAMR_BUILD_MINI_LOADER 0)
  set(WAMR_BUILD_EXCE_HANDLING 0)
  set(WAMR_BUILD_GC 0)
  set(WAMR_BUILD_DEBUG_INTERP 0)
  include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
  add_library(vmlib STATIC ${WAMR_RUNTIME_LIB_SOURCE})
  target_compile_options(vmlib PRIVATE -w)
  if(WAMR_BLOCK_CACHE GREATER 0)
    target_compile_definitions(vmlib PUBLIC BLOCK_ADDR_CACHE_SIZE=${WAMR_BLOCK_CACHE})
  endif()
  set(BACKEND_SRC backend_wamr.c wamr_host.c frame_host.c)
  set(BACKEND_LIBS vmlib)
  if(WAMR_FAST_INTERP EQUAL 1)
    set(FLAVOUR wamr-fast)
  else()
    set(FLAVOUR wamr-classic)
  endif()
endif()

function(sw_exe name src)
  add_executable(${name} ${src} mock.c ${BACKEND_SRC})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${S0}/core/src)
  if(SW_BACKEND STREQUAL "wamr")
    target_include_directories(${name} PRIVATE ${WAMR_ROOT_DIR}/core/iwasm/include)
  endif()
  target_compile_definitions(${name} PRIVATE SW_WAMR_FLAVOUR="${FLAVOUR}")
  target_compile_options(${name} PRIVATE -Wall -Wextra)
  target_link_libraries(${name} ${BACKEND_LIBS} Threads::Threads m)
endfunction()

sw_exe(sw_test scenarios.c)
sw_exe(sw_bench bench.c)

# S0 bootstrap (pure-JS DOM layer) extracted by the existing script, read-only use.
add_custom_command(
  OUTPUT ${CMAKE_BINARY_DIR}/bootstrap.js
  COMMAND python3 -I ${S0}/scripts/extract-bootstrap.py ${CMAKE_BINARY_DIR}/bootstrap.js
  DEPENDS ${S0}/scripts/extract-bootstrap.py)
add_custom_target(bootstrap ALL DEPENDS ${CMAKE_BINARY_DIR}/bootstrap.js)
